\textbf{定义3.1 }集合\hspace{0pt}$X$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊囿结构是一个映射\hspace{0pt}$\mathcal{B} :L ^ { X } \to M$\nobreak\hspace{0pt}，满足：

对\hspace{0pt}$\forall x\in X$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{B} \left ( { \mathcal{X}_{ \left \{ \begin{array}{l} x \end{array} \right \} } } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$A\le B$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{B} \left ( { B } \right )\le\allowbreak{} \mathcal{B} \left ( { A } \right )$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{B} \left ( { A\vee\allowbreak{} B } \right )\ge\allowbreak{} \mathcal{B} \left ( { A } \right )\wedge\allowbreak{} \mathcal{B} \left ( { B } \right )$\nobreak\hspace{0pt}.

则\hspace{0pt}$\left ( { X,\mathcal{B} } \right )$称为\hspace{0pt}$\left ( { L,M } \right )$-模糊囿空间，\hspace{0pt}$\mathcal{B}$中的元素成为有界的。

\textbf{注3.2} （1）一个\hspace{0pt}$\left ( { L,M } \right )$-模糊囿结构满足\hspace{0pt}$\mathcal{B} \left ( { \top _{ M } } \right )=\allowbreak{}\bot$\nobreak\hspace{0pt}，则称为\hspace{0pt}$\left ( { L,M } \right )$-模糊真囿结构；

（2）由定义3.1可知，对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{B} \left ( { A\vee\allowbreak{} B } \right )=\allowbreak{}\mathcal{B} \left ( { A } \right )\wedge\allowbreak{} \mathcal{B} \left ( { B } \right )$也成立。

\textbf{例3.3} 设\hspace{0pt}$\left ( { X,d } \right )$是一个经典的度量空间，对\hspace{0pt}$\forall x_{ 0 } \in X$\nobreak\hspace{0pt}，定义\hspace{0pt}$\mathcal{B} :\left[ 0,1 \right] ^ { X } \to\allowbreak{} \left[ 0,1 \right]$\nobreak\hspace{0pt}，
$$\mathcal{B} \left ( { A } \right )=\bigvee_{ r>0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } A'\left ( { x } \right )\text{，}$$
则\hspace{0pt}$\left ( { X,\mathcal{B} } \right )$称为\hspace{0pt}$\left ( { L,M } \right )$-模糊囿空间。

证明 （1）对\hspace{0pt}$\forall y\in X$\nobreak\hspace{0pt}，取\hspace{0pt}$r_{ 0 } =\allowbreak{}d\left ( { y,x_{ 0 } } \right )+1$\nobreak\hspace{0pt}，满足对\hspace{0pt}$\forall x\in X$\nobreak\hspace{0pt}，\hspace{0pt}$d\left ( { x,x_{ 0 } } \right )>r_{ 0 }$\nobreak\hspace{0pt}，则\hspace{0pt}$x\ne y$\nobreak\hspace{0pt}，故\hspace{0pt}$\mathcal{X}_{ \left \{ \begin{array}{l} y \end{array} \right \} } \left ( { x } \right )=\allowbreak{}0$\nobreak\hspace{0pt}，则\hspace{0pt}$\mathcal{X}_{ \left \{ \begin{array}{l} y \end{array} \right \} } '\left ( { x } \right )=\allowbreak{}1$\nobreak\hspace{0pt}，即\hspace{0pt}$\bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } \mathcal{X}_{ \left \{ \begin{array}{l} y \end{array} \right \} } '\left ( { x } \right )=\allowbreak{}1$\nobreak\hspace{0pt}，
\begin{eqnarray*} \mathcal{B} \left ( { \mathcal{X}_{ \left \{ \begin{array}{l} y \end{array} \right \} } } \right ) \\  &=& \bigvee_{ r>0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } \mathcal{X}_{ \left \{ \begin{array}{l} y \end{array} \right \} } '\left ( { x } \right )=1\text{；} \end{eqnarray*}


（2）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$A\le B$\nobreak\hspace{0pt}，则对\hspace{0pt}$\forall x\in X$\nobreak\hspace{0pt}，\hspace{0pt}$A\left ( { x } \right )\le\allowbreak{} B\left ( { x } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$A'\left ( { x } \right )\ge\allowbreak{} B'\left ( { x } \right )$\nobreak\hspace{0pt}，所以对给定的\hspace{0pt}$r>0$\nobreak\hspace{0pt}，\hspace{0pt}$\bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } A'\left ( { x } \right )\ge\allowbreak{} \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } B'\left ( { x } \right )$\nobreak\hspace{0pt}，故
\begin{eqnarray*} \mathcal{B} \left ( { A } \right ) &=& \bigvee_{ r>0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } A'\left ( { x } \right ) \\  &\ge& \bigvee_{ r>0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } B'\left ( { x } \right )=\mathcal{B} \left ( { B } \right )\text{；} \end{eqnarray*}
（3）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，由于
\begin{eqnarray*} \mathcal{B} \left ( { A } \right )\wedge \mathcal{B} \left ( { B } \right ) &=& \left ( { \bigvee_{ r_{ 1 } >0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 1 } } A'\left ( { x } \right ) } \right ) \\ \wedge \left ( { \bigvee_{ r_{ 2 } >0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 2 } } B'\left ( { x } \right ) } \right ) \\  &=& \bigvee_{ r_{ 1 } >0 } \bigvee_{ r_{ 2 } >0 } \left ( { \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 1 } } A'\left ( { x } \right )\wedge \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 2 } } B'\left ( { x } \right ) } \right ) \end{eqnarray*}
取\hspace{0pt}$r=\allowbreak{}max\left \{ \begin{array}{l} r_{ 1 } ,r_{ 2 } \end{array} \right \}$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 1 } } A'\left ( { x } \right )\wedge \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 2 } } B'\left ( { x } \right ) \\  &\le& \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } A'\left ( { x } \right )\wedge \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } B'\left ( { x } \right ) \\  &\le& \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } \left ( { A'\left ( { x } \right )\wedge B'\left ( { x } \right ) } \right ) \\  &=& \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } \left ( { A\vee B } \right )'\left ( { x } \right ) \end{eqnarray*}
故
\begin{eqnarray*} \mathcal{B} \left ( { A } \right )\wedge \mathcal{B} \left ( { B } \right ) &=& \bigvee_{ r_{ 1 } >0 } \bigvee_{ r_{ 2 } >0 } \left ( { \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 1 } } A'\left ( { x } \right )\wedge \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r_{ 2 } } B'\left ( { x } \right ) } \right ) \\  &\le& \bigvee_{ r>0 } \bigwedge_{ d\left ( { x,x_{ 0 } } \right )>r } \left ( { A\vee B } \right )'\left ( { x } \right )=\mathcal{B} \left ( { A\vee B } \right )\text{。} \end{eqnarray*}


\textbf{定义3.}\textbf{4} \hspace{0pt}$\left ( { X,\mathcal{B} } \right )$是一个\hspace{0pt}$\left ( { L,M } \right )$-模糊囿空间，定义映射\hspace{0pt}$\delta :L ^ { X } \times L ^ { X } \to\allowbreak{} M$是\hspace{0pt}$X$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近，若它满足：

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A,B } \right )=\allowbreak{}\delta \left ( { B,A } \right )$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A,B } \right )\le\allowbreak{} \mathcal{B} '\left ( { A } \right )\wedge\allowbreak{} \mathcal{B} '\left ( { B } \right )$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B,C\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A\vee\allowbreak{} B,C } \right )\le\allowbreak{} \delta \left ( { A,C } \right )\vee\allowbreak{} \delta \left ( { B,C } \right )$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A,B } \right )\ge\allowbreak{} \bigwedge_{ D\in\allowbreak{} L ^ { X } } \left ( { \delta \left ( { A,D } \right )\vee\allowbreak{} \delta \left ( { D',B } \right ) } \right )$\nobreak\hspace{0pt}；

对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta '\left ( { A,B } \right )\le\allowbreak{} \mathcal{B} \left ( { A\wedge\allowbreak{} B } \right )$\nobreak\hspace{0pt}.

则\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$称为\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间。

\textbf{注3.5}  若\hspace{0pt}$L=\allowbreak{}\left \{ \begin{array}{l} 0,1 \end{array} \right \}$\nobreak\hspace{0pt}，\hspace{0pt}$M$保留为一般的完全分配格，则退化为石老师定义的L-值粗邻近；若\hspace{0pt}$L,M=\allowbreak{}\left \{ \begin{array}{l} 0,1 \end{array} \right \}$\nobreak\hspace{0pt}，则退化为经典意义下的粗邻近

\textbf{注3.6}  设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，若\hspace{0pt}$\mathcal{B} \left ( { A } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}，则对\hspace{0pt}$\forall B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A,B } \right )=\allowbreak{}\bot$\nobreak\hspace{0pt}；特别地，若\hspace{0pt}$\mathcal{B} \left ( { \top _{ M } } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta$恒为\hspace{0pt}$\bot$\nobreak\hspace{0pt}。

\textbf{例3.7}  设\hspace{0pt}$\mathcal{B} :L ^ { X } \to M$为非空集合\hspace{0pt}$X$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊真囿结构，定义\hspace{0pt}$\delta _{ 1 } ,\delta _{ 2 } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$分别为，对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，
\begin{eqnarray*} \delta _{ 1 } \left ( { A,B } \right ) &=& \mathcal{B} '\left ( { A\wedge B } \right )\text{，}\delta _{ 2 } \left ( { A,B } \right )=\mathcal{B} '\left ( { A } \right ) \\ \wedge \mathcal{B} '\left ( { B } \right )\text{，} \end{eqnarray*}
则称\hspace{0pt}$\delta _{ 1 }$为\hspace{0pt}$\left ( { L,M } \right )$-模糊离散粗邻近，\hspace{0pt}$\delta _{ 2 }$为\hspace{0pt}$\left ( { L,M } \right )$-模糊平凡粗邻近。

\textbf{命题3.8} 设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，对\hspace{0pt}$\forall A,B,C,D\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$A\le C$\nobreak\hspace{0pt}，\hspace{0pt}$B\le D$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { A,B } \right )\le\allowbreak{} \delta \left ( { C,D } \right )$\nobreak\hspace{0pt}。

证明 由于\hspace{0pt}$A\le C$\nobreak\hspace{0pt}，\hspace{0pt}$B\le D$\nobreak\hspace{0pt}，则\hspace{0pt}$A\vee C=C$\nobreak\hspace{0pt}，\hspace{0pt}$B\vee D=D$\nobreak\hspace{0pt}，故
$$\delta \left ( { C,D } \right )=\delta \left ( { A\vee C,D } \right )=\delta \left ( { A,D } \right )\vee \delta \left ( { C,D } \right )\text{，}$$
所以\hspace{0pt}$\delta \left ( { A,D } \right )\le\allowbreak{} \delta \left ( { C,D } \right )$\nobreak\hspace{0pt}，同理可得\hspace{0pt}$\delta \left ( { A,B } \right )\le\allowbreak{} \delta \left ( { A,D } \right )$\nobreak\hspace{0pt}，故\hspace{0pt}$\delta \left ( { A,B } \right )\le\allowbreak{} \delta \left ( { C,D } \right )$\nobreak\hspace{0pt}。

\textbf{命题3.9} 设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，有
$$\delta \left ( { A,B } \right )\le \bigwedge_{ D\in L ^ { X } } \left ( { \delta \left ( { A,D } \right )\vee \delta \left ( { D',B } \right ) } \right )\text{。}$$
证明 对\hspace{0pt}$\forall D\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$B\le\allowbreak{} D\vee\allowbreak{} \left ( { B\wedge\allowbreak{} D' } \right )$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \delta \left ( { A,B } \right ) &\le& \delta \left ( { A,D\vee \left ( { B\wedge D' } \right ) } \right )=\delta \left ( { A,D } \right ) \\ \vee \delta \left ( { A,B\wedge D' } \right )\text{，} \end{eqnarray*}
而\hspace{0pt}$\delta '\left ( { B,D' } \right )\le\allowbreak{} \mathcal{B} \left ( { B\wedge\allowbreak{} D' } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { B,D' } \right )\ge\allowbreak{} \mathcal{B} '\left ( { B\wedge\allowbreak{} D' } \right )$\nobreak\hspace{0pt}，所以
\begin{eqnarray*} \delta \left ( { A,B\wedge D' } \right ) &\le& \mathcal{B} '\left ( { A } \right )\wedge \mathcal{B} '\left ( { B\wedge D' } \right ) \\  &\le& \mathcal{B} '\left ( { B\wedge D' } \right )\le \delta \left ( { B,D' } \right )\text{，} \end{eqnarray*}
故\hspace{0pt}$\delta \left ( { A,B } \right )\le\allowbreak{} \delta \left ( { A,D } \right )\vee\allowbreak{} \delta \left ( { D',B } \right )$\nobreak\hspace{0pt}，得证。

\textbf{定义3.10 }设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，定义映射\hspace{0pt}$N_{ \delta } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$\nobreak\hspace{0pt}，对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )=\allowbreak{}\left ( { \delta \left ( { A,B' } \right ) } \right )'$\nobreak\hspace{0pt}，\hspace{0pt}$N_{ \delta }$称为由\hspace{0pt}$\delta$诱导的\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻域。

\textbf{定理3.11 }设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，\hspace{0pt}$N_{ \delta }$称为由\hspace{0pt}$\delta$诱导的\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻域，则：对\hspace{0pt}$\forall A,B,C\in L ^ { X }$

（1）\hspace{0pt}$\mathcal{B} \left ( { A } \right )\le\allowbreak{} N_{ \delta } \left ( { A,B } \right )$\nobreak\hspace{0pt}；

（2）\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\le\allowbreak{} \mathcal{B} \left ( { A\wedge\allowbreak{} B' } \right )$\nobreak\hspace{0pt}；

（3）若\hspace{0pt}$A\le B$\nobreak\hspace{0pt}，则\hspace{0pt}$N_{ \delta } \left ( { B,C } \right )\le\allowbreak{} N_{ \delta } \left ( { A,C } \right )$\nobreak\hspace{0pt}；

（4）若\hspace{0pt}$B\le C$\nobreak\hspace{0pt}，则\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\le\allowbreak{} N_{ \delta } \left ( { A,C } \right )$\nobreak\hspace{0pt}；

（5）\hspace{0pt}$\mathcal{B} \left ( { A\wedge\allowbreak{} B' } \right )\wedge\allowbreak{} N_{ \delta } \left ( { B,C } \right )\le\allowbreak{} N_{ \delta } \left ( { A,C } \right )$\nobreak\hspace{0pt}；

（6）\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\wedge\allowbreak{} N_{ \delta } \left ( { B,C } \right )\le\allowbreak{} N_{ \delta } \left ( { A,C } \right )$\nobreak\hspace{0pt}；

（7）\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\wedge\allowbreak{} N_{ \delta } \left ( { A,B' } \right )\le\allowbreak{} \mathcal{B} \left ( { A } \right )$\nobreak\hspace{0pt}；

（8）\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )=\allowbreak{}N_{ \delta } \left ( { B',A' } \right )$\nobreak\hspace{0pt}。

证明 （1）对\hspace{0pt}$\forall A,B'\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$\delta \left ( { A,B' } \right )\le\allowbreak{} \mathcal{B} '\left ( { A } \right )\wedge\allowbreak{} \mathcal{B} '\left ( { B' } \right )$\nobreak\hspace{0pt}，所以\hspace{0pt}$\delta \left ( { A,B' } \right )\le\allowbreak{} \mathcal{B} '\left ( { A } \right )$\nobreak\hspace{0pt}，故
$$N_{ \delta } \left ( { A,B } \right )=\left ( { \delta \left ( { A,B' } \right ) } \right )'\ge \mathcal{B} \left ( { A } \right )\text{；}$$
（2）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$\delta '\left ( { A,B } \right )\le\allowbreak{} \mathcal{B} \left ( { A\wedge\allowbreak{} B } \right )$\nobreak\hspace{0pt}，则\newline \hspace{0pt}$N_{ \delta } \left ( { A,B } \right )=\allowbreak{}\delta '\left ( { A,B' } \right )\le\allowbreak{} \mathcal{B} \left ( { A\wedge\allowbreak{} B' } \right )$\nobreak\hspace{0pt}；

（3）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$A\le B$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { A,C' } \right )\le\allowbreak{} \delta \left ( { B,C' } \right )$\nobreak\hspace{0pt}，故
$$N_{ \delta } \left ( { A,C } \right )=\delta '\left ( { A,C' } \right )\ge \delta '\left ( { B,C' } \right )=N_{ \delta } \left ( { B,C } \right )\text{；}$$
（4）与（3）同理；

（5）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$A\le\allowbreak{} B\vee\allowbreak{} \left ( { A\wedge\allowbreak{} B' } \right )$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \delta \left ( { A,C' } \right ) &\le& \delta \left ( { B\vee \left ( { A\wedge B' } \right ),C' } \right )=\delta \left ( { B,C' } \right ) \\ \vee \delta \left ( { A\wedge B',C' } \right ) \end{eqnarray*}
又
\begin{eqnarray*} \delta \left ( { A\wedge B',C' } \right ) &\le& \mathcal{B} '\left ( { A\wedge B' } \right )\wedge \mathcal{B} '\left ( { C' } \right ) \\  &\le& \mathcal{B} '\left ( { A\wedge B' } \right )\text{，} \end{eqnarray*}
则\hspace{0pt}$\delta \left ( { A,C' } \right )\le\allowbreak{} \delta \left ( { B,C' } \right )\vee\allowbreak{} \mathcal{B} '\left ( { A\wedge\allowbreak{} B' } \right )$\nobreak\hspace{0pt}，即
\begin{eqnarray*} N_{ \delta } \left ( { A,C } \right ) &=& \delta '\left ( { A,C' } \right )\ge \delta '\left ( { B,C' } \right )\wedge \mathcal{B} \left ( { A\wedge B' } \right ) \\  &=& N_{ \delta } \left ( { B,C } \right )\wedge \mathcal{B} \left ( { A\wedge B' } \right )\text{；} \end{eqnarray*}
（6）对\hspace{0pt}$\forall A,B,C\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$\delta \left ( { A,C' } \right )=\allowbreak{}\bigwedge_{ E\in\allowbreak{} L ^ { X } } \left ( { \delta \left ( { A,E } \right )\vee\allowbreak{} \delta \left ( { E',C' } \right ) } \right )$\nobreak\hspace{0pt}，取\hspace{0pt}$E=B'$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { A,C' } \right )\le\allowbreak{} \delta \left ( { A,B' } \right )\vee\allowbreak{} \delta \left ( { B,C' } \right )$\nobreak\hspace{0pt}，故
\begin{eqnarray*} N_{ \delta } \left ( { A,C } \right ) &=& \delta '\left ( { A,C' } \right )\ge \delta '\left ( { A,B' } \right )\wedge \delta '\left ( { B,C' } \right ) \\  &=& N_{ \delta } \left ( { A,B } \right )\wedge N_{ \delta } \left ( { B,C } \right )\text{；} \end{eqnarray*}
（7）由于\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\wedge\allowbreak{} N_{ \delta } \left ( { B,A' } \right )\le\allowbreak{} N_{ \delta } \left ( { A,A' } \right )$\nobreak\hspace{0pt}，而\newline \hspace{0pt}$N_{ \delta } \left ( { B,A' } \right )=\allowbreak{}\delta '\left ( { B,A } \right )=\allowbreak{}\delta '\left ( { A,B } \right )$\nobreak\hspace{0pt}，同理\newline \hspace{0pt}$N_{ \delta } \left ( { A,B' } \right )=\allowbreak{}\delta '\left ( { A,B } \right )$\nobreak\hspace{0pt}，则\newline \hspace{0pt}$N_{ \delta } \left ( { A,B' } \right )=\allowbreak{}N_{ \delta } \left ( { B,A' } \right )$\nobreak\hspace{0pt}，故\newline \hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\wedge\allowbreak{} N_{ \delta } \left ( { A,B' } \right )\le\allowbreak{} N_{ \delta } \left ( { A,A' } \right )$\nobreak\hspace{0pt}，又由于\newline \hspace{0pt}$N_{ \delta } \left ( { A,A' } \right )=\allowbreak{}\delta '\left ( { A,A } \right )\le\allowbreak{} \mathcal{B} \left ( { A\wedge\allowbreak{} A } \right )=\allowbreak{}\mathcal{B} \left ( { A } \right )$\nobreak\hspace{0pt}，则\newline \hspace{0pt}$N_{ \delta } \left ( { A,B } \right )\wedge\allowbreak{} N_{ \delta } \left ( { A,B' } \right )\le\allowbreak{} \mathcal{B} \left ( { A } \right )$\nobreak\hspace{0pt}；

（8）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$N_{ \delta } \left ( { B',A' } \right )=\allowbreak{}\delta '\left ( { B',A } \right )=\allowbreak{}\delta '\left ( { A,B' } \right )=\allowbreak{}N_{ \delta } \left ( { A,B } \right )$\nobreak\hspace{0pt}。

\textbf{注3.1}\textbf{2}\textbf{ }设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，\hspace{0pt}$N_{ \delta } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$是由\hspace{0pt}$\delta$诱导的\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻域，则对\hspace{0pt}$\forall A,B,D\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{B} \left ( { D } \right )\wedge\allowbreak{} N_{ \delta } \left ( { A,B } \right )\le\allowbreak{} N_{ \delta } \left ( { A\vee\allowbreak{} D,B } \right )$\nobreak\hspace{0pt}。

\textbf{定理3.1}\textbf{3}\textbf{ }映射\hspace{0pt}$\mathcal{B} :L ^ { X } \to M$\nobreak\hspace{0pt}，是非空集合\hspace{0pt}$X$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊真囿结构，映射\hspace{0pt}$\delta :L ^ { X } \times L ^ { X } \to\allowbreak{} M$满足定义3.4里的（1）-（3），（5），设\hspace{0pt}$N_{ \delta } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$是由\hspace{0pt}$N_{ \delta } \left ( { A,B } \right )=\allowbreak{}\left ( { \delta \left ( { A,B' } \right ) } \right )'$定义的\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻域，则以下两条等价：

（1）对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { A,B } \right )\ge\allowbreak{} \bigwedge_{ D\in\allowbreak{} L ^ { X } } \left ( { \delta \left ( { A,D } \right )\vee\allowbreak{} \delta \left ( { D',B } \right ) } \right )$\nobreak\hspace{0pt}；（定义3.4的（4））

（2）推（1）：对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}，由于\hspace{0pt}$N_{ \delta } \left ( { A,B' } \right )\le\allowbreak{} \bigvee_{ C\in\allowbreak{} L ^ { X } } \left ( { N_{ \delta } \left ( { A,C } \right )\wedge\allowbreak{} N_{ \delta } \left ( { C,B' } \right ) } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta '\left ( { A,B } \right )\le\allowbreak{} \bigvee_{ C\in\allowbreak{} L ^ { X } } \left ( { \delta '\left ( { A,C' } \right )\vee\allowbreak{} \delta '\left ( { C,B } \right ) } \right )$\nobreak\hspace{0pt}，所以
\begin{eqnarray*} \delta \left ( { A,B } \right ) &\ge& \bigwedge_{ C\in L ^ { X } } \left ( { \delta \left ( { A,C' } \right )\vee \delta \left ( { C,B } \right ) } \right ) \\  &\ge& \bigwedge_{ D\in L ^ { X } } \left ( { \delta \left ( { A,D } \right )\vee \delta \left ( { D',B } \right ) } \right )\text{。} \end{eqnarray*}
\textbf{定义4.1}\textbf{ }设\hspace{0pt}$X$为一个非空集合。映射\hspace{0pt}$\varnothing :L ^ { X } \times L ^ { X } \to\allowbreak{} M$称为\hspace{0pt}$L ^ { X }$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊等价关系，若它满足以下条件：对\hspace{0pt}$\forall A,B,C\in L ^ { X }$\nobreak\hspace{0pt}，

（2）对称性：\hspace{0pt}$\varnothing \left ( { A,B } \right )=\allowbreak{}\varnothing \left ( { B,A } \right )$\nobreak\hspace{0pt}；

\begin{eqnarray*} \varnothing \left ( { A_{ 1 } \vee A_{ 2 } ,B } \right ) &\le& \bigvee_{ B_{ 1 } \vee B_{ 2 } =B } \varnothing \left ( { A_{ 1 } ,B_{ 1 } } \right ) \\ \wedge \varnothing \left ( { A_{ 2 } ,B_{ 2 } } \right )\text{。} \end{eqnarray*}
\textbf{例4.2 }设\hspace{0pt}$X$为一个非空集合，定义映射\hspace{0pt}$\varnothing _{ D } ,\varnothing _{ I } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$为，对\hspace{0pt}$\forall A,B\in L ^ { X }$\nobreak\hspace{0pt}：

\textbf{定理4.3}\textbf{ }设\hspace{0pt}$\left ( { X,\mathcal{B} ,\delta } \right )$为一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗邻近空间，定义映射\hspace{0pt}$\varnothing _{ \delta } :L ^ { X } \times L ^ { X } \to\allowbreak{} M$满足：
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right ) &=& \left ( { \bigwedge_{ A_{ 1 } \le A } \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,B } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ B_{ 1 } \le B } \mathcal{B} \left ( { B_{ 1 } } \right )\vee \delta \left ( { A,B_{ 1 } } \right ) } \right )\text{，} \end{eqnarray*}
则\hspace{0pt}$\varnothing _{ \delta }$称为\hspace{0pt}$X$上的一个伪\hspace{0pt}$\left ( { L,M } \right )$-模糊渐近相似性\textbf{。}

证明 先证\hspace{0pt}$\varnothing _{ \delta }$为\hspace{0pt}$L ^ { X }$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊等价关系：对\hspace{0pt}$\forall A,B,C,D\in L ^ { X }$\nobreak\hspace{0pt}，

（1）由于\hspace{0pt}$\delta \left ( { A,B } \right )=\allowbreak{}\delta \left ( { B,A } \right )$\nobreak\hspace{0pt}，所以显然\hspace{0pt}$\varnothing _{ \delta } \left ( { A,B } \right )=\allowbreak{}\varnothing _{ \delta } \left ( { B,A } \right )$\nobreak\hspace{0pt}；

（2）对\hspace{0pt}$\forall A_{ 1 } \le A$\nobreak\hspace{0pt}，由于\hspace{0pt}$\delta '\left ( { A_{ 1 } ,A } \right )\le\allowbreak{} \mathcal{B} \left ( { A_{ 1 } \wedge\allowbreak{} A } \right )=\allowbreak{}\mathcal{B} \left ( { A_{ 1 } } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { A_{ 1 } ,A } \right )\ge\allowbreak{} \mathcal{B} '\left ( { A_{ 1 } } \right )$\nobreak\hspace{0pt}，故
\begin{eqnarray*} \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,A } \right ) &\ge& \mathcal{B} \left ( { A_{ 1 } } \right ) \\ \vee \mathcal{B} '\left ( { A_{ 1 } } \right ) &=& \top\text{，} \end{eqnarray*}
同理，\hspace{0pt}$\mathcal{B} \left ( { A } \right )\vee\allowbreak{} \delta \left ( { A,A_{ 1 } } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}，则\hspace{0pt}$\varnothing _{ \delta } \left ( { A,A } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}；

（3）对\hspace{0pt}$\forall A_{ 1 } \le A$\nobreak\hspace{0pt}，由于\hspace{0pt}$B\le\allowbreak{} D\vee\allowbreak{} \left ( { B\wedge\allowbreak{} D' } \right )$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \delta \left ( { A_{ 1 } ,B } \right ) &\le& \delta \left ( { A_{ 1 } ,D\vee \left ( { B\wedge D' } \right ) } \right )=\delta \left ( { A_{ 1 } ,D } \right ) \\ \vee \delta \left ( { A_{ 1 } ,B\wedge D' } \right )\text{，} \end{eqnarray*}
又因为
\begin{eqnarray*} \delta \left ( { A_{ 1 } ,B\wedge D' } \right ) &\le& \mathcal{B} '\left ( { A_{ 1 } } \right )\wedge \mathcal{B} '\left ( { B\wedge D' } \right ) \\  &\le& \mathcal{B} '\left ( { B\wedge D' } \right )\text{，} \end{eqnarray*}
所以\hspace{0pt}$\delta \left ( { A_{ 1 } ,B } \right )\le\allowbreak{} \delta \left ( { A_{ 1 } ,D } \right )\vee\allowbreak{} \mathcal{B} '\left ( { B\wedge\allowbreak{} D' } \right )$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right ) &=& \left ( { \bigwedge_{ A_{ 1 } \le A } \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,B } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ B_{ 1 } \le B } \mathcal{B} \left ( { B_{ 1 } } \right )\vee \delta \left ( { A,B_{ 1 } } \right ) } \right ) \\  &\le& \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,B } \right )\le \mathcal{B} \left ( { A_{ 1 } } \right ) \\ \vee \delta \left ( { A_{ 1 } ,D } \right )\vee \mathcal{B} '\left ( { B\wedge D' } \right )\text{，} \end{eqnarray*}
由于\hspace{0pt}$B\wedge D'\le B$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { B,C } \right ) &=& \left ( { \bigwedge_{ B_{ 1 } \le B } \mathcal{B} \left ( { B_{ 1 } } \right )\vee \delta \left ( { B_{ 1 } ,C } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ C_{ 1 } \le C } \mathcal{B} \left ( { C_{ 1 } } \right )\vee \delta \left ( { B,C_{ 1 } } \right ) } \right ) \\  &\le& \mathcal{B} \left ( { B\wedge D' } \right )\vee \delta \left ( { B\wedge D',C } \right )\text{，} \end{eqnarray*}
而\hspace{0pt}$B\wedge D'\le D'$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { B\wedge\allowbreak{} D',C } \right )\le\allowbreak{} \delta \left ( { D',C } \right )$\nobreak\hspace{0pt}，故\hspace{0pt}$\varnothing _{ \delta } \left ( { B,C } \right )\le\allowbreak{} \mathcal{B} \left ( { B\wedge\allowbreak{} D' } \right )\vee\allowbreak{} \delta \left ( { D',C } \right )$\nobreak\hspace{0pt}，将上面得到的两个式子相交可得：
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { B,C } \right ) \\  &\le& \left ( { \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,D } \right )\vee \mathcal{B} '\left ( { B\wedge D' } \right ) } \right ) \\ \wedge \left ( { \mathcal{B} \left ( { B\wedge D' } \right )\vee \delta \left ( { D',C } \right ) } \right )\text{，} \end{eqnarray*}
故
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { B,C } \right ) &\le& \mathcal{B} \left ( { A_{ 1 } } \right ) \\ \vee \delta \left ( { A_{ 1 } ,D } \right )\vee \delta \left ( { D',C } \right )\text{，} \end{eqnarray*}
则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { B,C } \right ) &\le& \mathcal{B} \left ( { A_{ 1 } } \right ) \\ \vee \bigvee_{ D\in L ^ { X } } \left ( { \delta \left ( { A_{ 1 } ,D } \right )\vee \delta \left ( { D',C } \right ) } \right ) \\  &=& \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,C } \right )\text{，} \end{eqnarray*}
同理可得，都\hspace{0pt}$\forall C_{ 1 } \le C$\nobreak\hspace{0pt}，
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { B,C } \right ) &\le& \mathcal{B} \left ( { C_{ 1 } } \right ) \\ \vee \delta \left ( { A,C_{ 1 } } \right )\text{，} \end{eqnarray*}
则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { B,C } \right ) \\  &\le& \left ( { \bigwedge_{ A_{ 1 } \le A } \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,C } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ C_{ 1 } \le C } \mathcal{B} \left ( { C_{ 1 } } \right )\vee \delta \left ( { A,C_{ 1 } } \right ) } \right ) \\  &=& \varnothing _{ \delta } \left ( { A,C } \right )\text{，} \end{eqnarray*}
得证\hspace{0pt}$\varnothing _{ \delta }$为\hspace{0pt}$L ^ { X }$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊等价关系；

\quad 下证对\hspace{0pt}$\forall A,B,C,D\in L ^ { X }$\nobreak\hspace{0pt}，
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { C,D } \right ) \\  &\le& \varnothing _{ \delta } \left ( { A\vee C,B\vee D } \right )\text{：} \end{eqnarray*}


令\hspace{0pt}$A\vee C=U$\nobreak\hspace{0pt}，\hspace{0pt}$B\vee D=V$\nobreak\hspace{0pt}，对\hspace{0pt}$\forall U_{ 1 } \le U$\nobreak\hspace{0pt}，由于\hspace{0pt}$U_{ 1 } =\allowbreak{}\left ( { U_{ 1 } \wedge\allowbreak{} A } \right )\vee\allowbreak{} \left ( { U_{ 1 } \wedge\allowbreak{} C } \right )$\nobreak\hspace{0pt}，记\hspace{0pt}$U_{ A } =\allowbreak{}U_{ 1 } \wedge\allowbreak{} A\le\allowbreak{} A$\nobreak\hspace{0pt}，\hspace{0pt}$U_{ C } =\allowbreak{}U_{ 1 } \wedge\allowbreak{} C\le\allowbreak{} C$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right ) &=& \left ( { \bigwedge_{ A_{ 1 } \le A } \mathcal{B} \left ( { A_{ 1 } } \right )\vee \delta \left ( { A_{ 1 } ,B } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ B_{ 1 } \le B } \mathcal{B} \left ( { B_{ 1 } } \right )\vee \delta \left ( { A,B_{ 1 } } \right ) } \right ) \\  &\le& \mathcal{B} \left ( { U_{ A } } \right )\vee \delta \left ( { U_{ A } ,B } \right )\text{，} \end{eqnarray*}
\begin{eqnarray*} \varnothing _{ \delta } \left ( { C,D } \right ) &=& \left ( { \bigwedge_{ C_{ 1 } \le C } \mathcal{B} \left ( { C_{ 1 } } \right )\vee \delta \left ( { C_{ 1 } ,D } \right ) } \right ) \\ \wedge \left ( { \bigwedge_{ D_{ 1 } \le D } \mathcal{B} \left ( { D_{ 1 } } \right )\vee \delta \left ( { C,D_{ 1 } } \right ) } \right ) \\  &\le& \mathcal{B} \left ( { U_{ C } } \right )\vee \delta \left ( { U_{ C } ,D } \right )\text{，} \end{eqnarray*}
又由于\hspace{0pt}$B\le V$\nobreak\hspace{0pt}，\hspace{0pt}$D\le V$\nobreak\hspace{0pt}，则\hspace{0pt}$\delta \left ( { U_{ A } ,B } \right )\le\allowbreak{} \delta \left ( { U_{ A } ,V } \right )$\nobreak\hspace{0pt}，\hspace{0pt}$\delta \left ( { U_{ C } ,D } \right )\le\allowbreak{} \delta \left ( { U_{ C } ,V } \right )$\nobreak\hspace{0pt}，两式相交可得
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { C,D } \right ) \\  &\le& \left ( { \mathcal{B} \left ( { U_{ A } } \right )\vee \delta \left ( { U_{ A } ,V } \right ) } \right ) \\ \wedge \left ( { \mathcal{B} \left ( { U_{ C } } \right )\vee \delta \left ( { U_{ C } ,V } \right ) } \right ) \\  &=& \left ( { \mathcal{B} \left ( { U_{ A } } \right )\wedge \mathcal{B} \left ( { U_{ C } } \right ) } \right ) \\ \vee \left ( { \mathcal{B} \left ( { U_{ A } } \right )\wedge \delta \left ( { U_{ C } ,V } \right ) } \right ) \\ \vee \left ( { \delta \left ( { U_{ A } ,V } \right )\wedge \mathcal{B} \left ( { U_{ C } } \right ) } \right ) \\ \vee \left ( { \delta \left ( { U_{ A } ,V } \right )\wedge \delta \left ( { U_{ C } ,V } \right ) } \right ) \\  &\le& \left ( { \mathcal{B} \left ( { U_{ A } } \right )\wedge \mathcal{B} \left ( { U_{ C } } \right ) } \right )\vee \delta \left ( { U_{ C } ,V } \right ) \\ \vee \delta \left ( { U_{ A } ,V } \right ) \end{eqnarray*}
又因为
\begin{eqnarray*} \mathcal{B} \left ( { U_{ A } } \right )\wedge \mathcal{B} \left ( { U_{ C } } \right ) &=& \mathcal{B} \left ( { U_{ A } \vee U_{ C } } \right ) \\  &=& \mathcal{B} \left ( { U_{ 1 } } \right )\text{，}\delta \left ( { U_{ C } ,V } \right )\vee \delta \left ( { U_{ A } ,V } \right ) \\  &=& \delta \left ( { U_{ C } \vee U_{ A } ,V } \right )=\delta \left ( { U_{ 1 } ,V } \right )\text{，} \end{eqnarray*}
则
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { C,D } \right ) &\le& \mathcal{B} \left ( { U_{ 1 } } \right ) \\ \vee \delta \left ( { U_{ 1 } ,V } \right )\text{，} \end{eqnarray*}
同理对\hspace{0pt}$\forall V_{ 1 } \le V$\nobreak\hspace{0pt}，
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { C,D } \right ) &\le& \mathcal{B} \left ( { V_{ 1 } } \right ) \\ \vee \delta \left ( { U,V_{ 1 } } \right )\text{，} \end{eqnarray*}
故得证
\begin{eqnarray*} \varnothing _{ \delta } \left ( { A,B } \right )\wedge \varnothing _{ \delta } \left ( { C,D } \right ) &\le& \varnothing _{ \delta } \left ( { U,V } \right ) \\  &=& \varnothing _{ \delta } \left ( { A\vee C,B\vee D } \right )\text{。} \end{eqnarray*}
\textbf{定理4.4}\textbf{ }\hspace{0pt}$L ^ { X }$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊等价关系\hspace{0pt}$\varnothing$为\hspace{0pt}$X$上的一个伪\hspace{0pt}$\left ( { L,M } \right )$-模糊渐近相似性，定义\hspace{0pt}$\mathcal{C}_{ \varnothing } :L ^ { X\times X } \to\allowbreak{} M$满足，对\hspace{0pt}$\forall E\in\allowbreak{} L ^ { X\times X }$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { E } \right )=\allowbreak{}\bigwedge_{ F\le\allowbreak{} E } \varnothing \left ( { D_{ F } ,R_{ F } } \right )$\nobreak\hspace{0pt}，其中\hspace{0pt}$D_{ F } \left ( { x } \right )=\allowbreak{}\bigvee_{ y\in\allowbreak{} X } F\left ( { x,y } \right )$\nobreak\hspace{0pt}，\hspace{0pt}$R_{ F } \left ( { y } \right )=\allowbreak{}\bigvee_{ x\in\allowbreak{} X } F\left ( { x,y } \right )$\nobreak\hspace{0pt}，则称\hspace{0pt}$\mathcal{C}_{ \varnothing }$为\hspace{0pt}$L ^ { X }$上的一个\hspace{0pt}$\left ( { L,M } \right )$-模糊粗结构。

证明 （1）由于\hspace{0pt}$\Delta x=\Delta x ^ { -1 }$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { \Delta x } \right ) &=& \bigwedge_{ F\le \Delta x ^ { -1 } } \varnothing \left ( { D_{ F } ,R_{ F } } \right ) \\  &=& \bigwedge_{ F\le \Delta x } \varnothing \left ( { D_{ F } ,R_{ F } } \right )\text{，} \end{eqnarray*}
对\hspace{0pt}$\forall F\le \Delta x$\nobreak\hspace{0pt}，则\hspace{0pt}$F\left ( { x,y } \right )\le\allowbreak{} \Delta x\left ( { x,y } \right )$\nobreak\hspace{0pt}，当\hspace{0pt}$x\ne y$时，\hspace{0pt}$\Delta x\left ( { x,y } \right )=\allowbreak{}\bot$\nobreak\hspace{0pt}，则\hspace{0pt}$F\left ( { x,y } \right )=\allowbreak{}\bot$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { \Delta x } \right )=\allowbreak{}\varnothing \left ( { \bot ,\bot } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}；当\hspace{0pt}$x=y$时，\hspace{0pt}$D_{ F } \left ( { x } \right )=\allowbreak{}\bigvee_{ y\in\allowbreak{} X } F\left ( { x,y } \right )=\allowbreak{}F\left ( { x,x } \right )$\nobreak\hspace{0pt}，\hspace{0pt}$R_{ F } \left ( { y } \right )=\allowbreak{}\bigvee_{ x\in\allowbreak{} X } F\left ( { x,y } \right )=\allowbreak{}F\left ( { x,x } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$\varnothing \left ( { D_{ F } ,R_{ F } } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}，故\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { \Delta x } \right )=\allowbreak{}\top$\nobreak\hspace{0pt}；

（2）对\hspace{0pt}$\forall E,G\in\allowbreak{} L ^ { X\times X }$\nobreak\hspace{0pt}，\hspace{0pt}$E\le G$\nobreak\hspace{0pt}，即\hspace{0pt}$\forall F\le E$\nobreak\hspace{0pt}，一定有\hspace{0pt}$F\le G$\nobreak\hspace{0pt}，显然\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { E } \right )\ge\allowbreak{} \mathcal{C}_{ \varnothing } \left ( { G } \right )$\nobreak\hspace{0pt}；

（3）由于\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { E\vee\allowbreak{} G } \right )=\allowbreak{}\bigwedge_{ F\le\allowbreak{} E\vee\allowbreak{} G } \varnothing \left ( { D_{ F } ,R_{ F } } \right )$\nobreak\hspace{0pt}，对\hspace{0pt}$\forall F\le E\vee G$\nobreak\hspace{0pt}，令\hspace{0pt}$F_{ E } =F\wedge E$\nobreak\hspace{0pt}，\hspace{0pt}$F_{ G } =F\wedge G$\nobreak\hspace{0pt}，则\hspace{0pt}$F_{ E } \le E$\nobreak\hspace{0pt}，\hspace{0pt}$F_{ G } \le G$\nobreak\hspace{0pt}，且\hspace{0pt}$F=\allowbreak{}F\wedge\allowbreak{} \left ( { E\vee\allowbreak{} G } \right )=\allowbreak{}F_{ E } \vee\allowbreak{} F_{ G }$\nobreak\hspace{0pt}，则\hspace{0pt}$D_{ F } =\allowbreak{}D_{ F_{ E } \vee\allowbreak{} F_{ G } } =\allowbreak{}D_{ F_{ E } } \vee\allowbreak{} D_{ F_{ G } }$\nobreak\hspace{0pt}，\hspace{0pt}$R_{ F } =\allowbreak{}R_{ F_{ E } \vee\allowbreak{} F_{ G } } =\allowbreak{}R_{ F_{ E } } \vee\allowbreak{} R_{ F_{ G } }$\nobreak\hspace{0pt}，故
\begin{eqnarray*} \varnothing \left ( { D_{ F_{ E } } ,R_{ F_{ E } } } \right )\wedge \varnothing \left ( { D_{ F_{ G } } ,R_{ F_{ G } } } \right ) \\  &\le& \varnothing \left ( { D_{ F_{ E } } \vee D_{ F_{ G } } ,R_{ F_{ E } } \vee R_{ F_{ G } } } \right )=\varnothing \left ( { D_{ F } ,R_{ F } } \right )\text{，} \end{eqnarray*}
由于\hspace{0pt}$F_{ E } \le E$\nobreak\hspace{0pt}，\hspace{0pt}$F_{ G } \le G$\nobreak\hspace{0pt}，则\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { E } \right )\le\allowbreak{} \varnothing \left ( { D_{ F_{ E } } ,R_{ F_{ E } } } \right )$\nobreak\hspace{0pt}，\hspace{0pt}$\mathcal{C}_{ \varnothing } \left ( { G } \right )\le\allowbreak{} \varnothing \left ( { D_{ F_{ G } } ,R_{ F_{ G } } } \right )$\nobreak\hspace{0pt}，故
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E } \right )\wedge \mathcal{C}_{ \varnothing } \left ( { G } \right ) \\  &\le& \varnothing \left ( { D_{ F_{ E } } ,R_{ F_{ E } } } \right )\wedge \varnothing \left ( { D_{ F_{ G } } ,R_{ F_{ G } } } \right ) \\  &\le& \varnothing \left ( { D_{ F } ,R_{ F } } \right )\text{，} \end{eqnarray*}
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E } \right )\wedge \mathcal{C}_{ \varnothing } \left ( { G } \right ) \\  &\le& \bigwedge_{ F\le E\vee G } \varnothing \left ( { D_{ F } ,R_{ F } } \right )=\mathcal{C}_{ \varnothing } \left ( { E\vee G } \right )\text{；} \end{eqnarray*}
（4）对\hspace{0pt}$\forall F\le E\circ G$\nobreak\hspace{0pt}，即\hspace{0pt}$F\left ( { x,y } \right )\le\allowbreak{} \bigvee_{ z\in\allowbreak{} X } \left ( { E\left ( { x,z } \right )\wedge\allowbreak{} G\left ( { z,y } \right ) } \right )$\nobreak\hspace{0pt}，令
\begin{eqnarray*} F_{ E } \left ( { x,z } \right ) &=& \bigvee_{ y\in X } \left ( { F\left ( { x,y } \right )\wedge E\left ( { x,z } \right )\wedge G\left ( { z,y } \right ) } \right )\text{，}F_{ G } \left ( { z,y } \right ) \\  &=& \bigvee_{ x\in X } \left ( { F\left ( { x,y } \right )\wedge E\left ( { x,z } \right )\wedge G\left ( { z,y } \right ) } \right )\text{，} \end{eqnarray*}
显然\hspace{0pt}$F_{ E } \le E$\nobreak\hspace{0pt}，\hspace{0pt}$F_{ G } \le G$\nobreak\hspace{0pt}，又因为
\begin{eqnarray*} D_{ F_{ E } } \left ( { x } \right ) &=& \bigvee_{ z\in X } F_{ E } \left ( { x,z } \right ) \\  &=& \bigvee_{ z,y\in X } \left ( { F\left ( { x,y } \right )\wedge E\left ( { x,z } \right )\wedge G\left ( { z,y } \right ) } \right ) \\  &=& \bigvee_{ y\in X } F\left ( { x,y } \right )=D_{ F } \left ( { x } \right )\text{，} \end{eqnarray*}
所以\hspace{0pt}$D_{ F_{ E } } =D_{ F }$\nobreak\hspace{0pt}，同理可得\hspace{0pt}$R_{ F_{ G } } =R_{ F }$\nobreak\hspace{0pt}，又因为
\begin{eqnarray*} R_{ F_{ E } } \left ( { z } \right ) &=& \bigvee_{ x\in X } F_{ E } \left ( { x,z } \right ) \\  &=& \bigvee_{ x,y\in X } \left ( { F\left ( { x,y } \right )\wedge E\left ( { x,z } \right )\wedge G\left ( { z,y } \right ) } \right )\text{，} \end{eqnarray*}
\begin{eqnarray*} D_{ F_{ G } } \left ( { z } \right ) &=& \bigvee_{ y\in X } F_{ G } \left ( { z,y } \right ) \\  &=& \bigvee_{ x,y\in X } \left ( { F\left ( { x,y } \right )\wedge E\left ( { x,z } \right )\wedge G\left ( { z,y } \right ) } \right )\text{，} \end{eqnarray*}
则\hspace{0pt}$R_{ F_{ E } } =D_{ F_{ G } }$\nobreak\hspace{0pt}，由于\hspace{0pt}$F_{ E } \le E$及\hspace{0pt}$F_{ G } \le G$\nobreak\hspace{0pt}，则
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E } \right ) &\le& \varnothing \left ( { D_{ F_{ E } } ,R_{ F_{ E } } } \right ) \\  &=& \varnothing \left ( { D_{ F } ,R_{ F_{ E } } } \right )\text{，}\mathcal{C}_{ \varnothing } \left ( { G } \right ) \\  &\le& \varnothing \left ( { D_{ F_{ G } } ,R_{ F_{ G } } } \right )=\varnothing \left ( { D_{ F_{ G } } ,R_{ F } } \right )\text{，} \end{eqnarray*}
故
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E } \right )\wedge \mathcal{C}_{ \varnothing } \left ( { G } \right ) \\  &\le& \varnothing \left ( { D_{ F } ,R_{ F_{ E } } } \right )\wedge \varnothing \left ( { D_{ F_{ G } } ,R_{ F } } \right ) \\  &\le& \varnothing \left ( { D_{ F } ,R_{ F } } \right )\text{，} \end{eqnarray*}
则
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E } \right )\wedge \mathcal{C}_{ \varnothing } \left ( { G } \right ) \\  &\le& \bigwedge_{ F\le E\circ G } \varnothing \left ( { D_{ F } ,R_{ F } } \right )=\mathcal{C}_{ \varnothing } \left ( { E\circ G } \right )\text{；} \end{eqnarray*}
（6）对\hspace{0pt}$F\leq E^{-1}$\nobreak\hspace{0pt}，有\hspace{0pt}$F\left ( { x,y } \right )\le\allowbreak{} E ^ { -1 } \left ( { x,y } \right )$\nobreak\hspace{0pt}，则\hspace{0pt}$F ^ { -1 } \left ( { y,x } \right )\le\allowbreak{} E ^ { -1 } \left ( { x,y } \right )=\allowbreak{}E\left ( { y,x } \right )$\nobreak\hspace{0pt}，即\hspace{0pt}$F ^ { -1 } \le E$\nobreak\hspace{0pt}，而
\begin{eqnarray*} R_{ F ^ { -1 } } \left ( { y } \right ) &=& \bigvee_{ x\in X } F ^ { -1 } \left ( { x,y } \right )=\bigvee_{ x\in X } F\left ( { y,x } \right ) \\  &=& D_{ F } \left ( { x } \right )\text{，} \end{eqnarray*}
则
\begin{eqnarray*} \varnothing \left ( { D_{ F } ,R_{ F } } \right ) &=& \varnothing \left ( { R_{ F ^ { -1 } } ,D_{ F ^ { -1 } } } \right ) \\  &=& \varnothing \left ( { D_{ F ^ { -1 } } ,R_{ F ^ { -1 } } } \right )\text{，} \end{eqnarray*}
故
\begin{eqnarray*} \mathcal{C}_{ \varnothing } \left ( { E ^ { -1 } } \right ) &=& \bigwedge_{ F\le E ^ { -1 } } \varnothing \left ( { D_{ F } ,R_{ F } } \right ) \\  &=& \bigwedge_{ F ^ { -1 } \le E } \varnothing \left ( { D_{ F ^ { -1 } } ,R_{ F ^ { -1 } } } \right )=\mathcal{C}_{ \varnothing } \left ( { E } \right )\text{。} \end{eqnarray*}
